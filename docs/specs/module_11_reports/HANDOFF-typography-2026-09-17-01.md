```sql
select set_config('request.jwt.claims',
  json_build_object('email', u.email, 'role','authenticated')::text, true)
  from public.users u join public.permissions p on p.role_id=u.role_id
  join public.modules m on m.module_id=p.module_id

set local role authenticated;   -- רק אחרי ההזרקה
```
