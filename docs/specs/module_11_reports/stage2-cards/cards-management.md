**המדידות, מילה-במילה** *(‏Supabase MCP `execute_sql`, פרויקט `yfeovxppnfoafmfbdfvh`, קריאה-בלבד, <span dir="ltr">10/09/2026</span>)*:

```sql
-- ארבעת האריחים + השוואת-YTD (מחזיר גם 2024 כבקרה)
with pop as (
  select p.project_id,
    case when p.final_event_date between date'2026-01-01' and date'2026-09-10' then '2026'
         when p.final_event_date between date'2025-01-01' and date'2025-09-10' then '2025'
         when p.final_event_date between date'2024-01-01' and date'2024-09-10' then '2024' end ytd
  from projects p
  where p.project_status in ('finished','awaiting_payment','awaiting_invoice','event_finished')
    and p.final_event_date is not null)
select ytd, count(*) n, round(sum(fm.revenue))::bigint rev, round(sum(fm.gross_profit))::bigint prof,
       round(100.0*sum(fm.gross_profit)/nullif(sum(fm.revenue),0),1) margin
from pop join lateral finance_project_money(pop.project_id) fm on true   -- 🔴 set-returning ⇒ cross join lateral
where ytd is not null group by ytd order by ytd;
-- ⇒ 2024: 114 · 949,252 · 525,480 · 55.4  |  2025: 178 · 1,385,815 · 774,665 · 55.9
--   2026: 236 · 1,922,441 · 1,118,098 · 58.2
```

```sql
-- נתח 5 הגדולים (כל הזמנים). החלף 2026-09-10 ב-2025-09-10 לקבלת ההשוואה.
with pop as (select p.project_id, p.customer_id from projects p
  where p.project_status in ('finished','awaiting_payment','awaiting_invoice','event_finished')
    and p.final_event_date is not null and p.final_event_date <= date'2026-09-10'),
rev as (select pop.customer_id, sum(fm.revenue) rev
  from pop join lateral finance_project_money(pop.project_id) fm on true group by 1),
ranked as (select *, row_number() over (order by rev desc) rn, sum(rev) over () total, count(*) over () ncust from rev)
select ncust, round(total)::bigint total, round(sum(rev) filter (where rn<=5))::bigint top5,
       round(100.0*sum(rev) filter (where rn<=5)/max(total),1) pct
from ranked group by ncust, total;
-- ⇒ היום: 55 · 5,793,488 · 2,637,121 · 45.5   |   לפני שנה: 47 · 3,011,462 · 1,437,380 · 47.7
```

```sql
-- שלוש השנים + שעות + שכר + בונוסים, בשאילתה אחת
with pop as (
  select p.project_id, extract(year from p.final_event_date)::int yr from projects p
  where p.project_status in ('finished','awaiting_payment','awaiting_invoice','event_finished')
    and p.final_event_date is not null and p.final_event_date <= date'2026-09-10'),
rev as (select yr, count(*) n, sum(fm.revenue) revenue, sum(fm.gross_profit) profit
        from pop join lateral finance_project_money(pop.project_id) fm on true group by yr),
lab as (select pop.yr, sum(a.actual_hours) hrs,
               sum(a.actual_hours*a.hourly_rate_snapshot) wage,
               sum(coalesce(a.personal_bonus,0)) bonus,
               sum(coalesce(a.travel_amount,0)) travel
        from pop join assignments a on a.project_id = pop.project_id      -- 🔴 מפתח (project_id, hostess_id,
        where a.actual_hours is not null and a.actual_hours > 0           --    assignment_number); הצירוף כאן
        group by pop.yr)                                                  --    הוא projects→assignments 1:N בלבד
select r.yr, r.n, round(r.revenue)::bigint rev, round(r.profit)::bigint prof,
       round(100.0*r.profit/r.revenue,1) margin, round(l.hrs,1) hours,
       round(r.revenue/l.hrs,1) price_per_hour,
       round((l.wage+l.bonus)/l.hrs,1) cost_per_hour        -- ⑧ G6: שכר+בונוסים, בלי נסיעות
from rev r join lab l on l.yr=r.yr order by r.yr;
```

```sql
    coalesce(q.applied_customer_discount,0)+coalesce(q.manual_discount,0) d
  from quotes q where q.quote_status='approved'),
tier as (select quote_id, case when d=0 then '0' when d<=5 then '1-5'

proj as (select t.b, p.project_id from tier t join projects p on p.quote_id=t.quote_id

    and p.final_event_date is not null and p.final_event_date <= date'2026-09-10'),
money as (select proj.b, count(*) np, sum(fm.revenue) rev, sum(fm.gross_profit) prof
  from proj join lateral finance_project_money(proj.project_id) fm on true group by 1)
select t.b, count(*) quotes_n, m.np, round(m.rev)::bigint rev,
       round(100.0*m.prof/nullif(m.rev,0),1) margin
from tier t left join money m on m.b=t.b group by t.b, m.np, m.rev, m.prof
order by case t.b when '0' then 1 when '1-5' then 2 when '6-10' then 3 else 4 end;
```
